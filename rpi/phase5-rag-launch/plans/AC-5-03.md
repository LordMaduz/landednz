## retrieval.ts implementation

```typescript
// apps/api/src/features/chat/rag/retrieval.ts
import { db } from '../../shared/db';
import { generateEmbedding } from '../../shared/ai/embeddings.client';
import { logger } from '../../shared/logger';

interface RetrievalInput {
  query: string;
  personaKey: string;
  visaType: string;
  city: string;
  taskContext: string | null;
}

interface RetrievedChunk {
  chunkKey: string;
  content: string;
  sourceUrl: string;
  sourceName: string;
  similarityScore: number;
}

interface RawChunkRow {
  chunkKey: string;
  content: string;
  sourceUrl: string;
  sourceName: string;
  personaKey: string | null;
  contentVersion: number;
  similarity: number;
}

export async function retrieveChunks(input: RetrievalInput): Promise<RetrievedChunk[]> {
  // Step 2: Generate query embedding
  const queryEmbedding = await generateEmbedding(input.query);
  const vectorLiteral = `[${queryEmbedding.join(',')}]`;

  // Step 1 (in SQL) + Step 3: Persona pre-filter (C07 — SQL WHERE, not post-filter) + semantic search
  const rawRows = await db.$queryRaw<RawChunkRow[]>`
    SELECT
      "chunkKey",
      content,
      "sourceUrl",
      "sourceName",
      "personaKey",
      "contentVersion",
      1 - (embedding <=> ${vectorLiteral}::vector) AS similarity
    FROM "KnowledgeBaseChunk"
    WHERE ("personaKey" = ${input.personaKey} OR "personaKey" IS NULL)
    ORDER BY embedding <=> ${vectorLiteral}::vector
    LIMIT 10
  `;

  // C10: BigInt replacer on all $queryRaw results
  const rows: RawChunkRow[] = JSON.parse(
    JSON.stringify(rawRows, (_, v) => (typeof v === 'bigint' ? Number(v) : v))
  );

  // Step 4: Re-rank — similarity (primary) + persona exactness (secondary) + contentVersion (tertiary)
  const reranked = [...rows].sort((a, b) => {
    const diff = b.similarity - a.similarity;
    if (Math.abs(diff) >= 0.01) return diff;
    const aExact = a.personaKey === input.personaKey ? 1 : 0;
    const bExact = b.personaKey === input.personaKey ? 1 : 0;
    if (aExact !== bExact) return bExact - aExact;
    return (b.contentVersion ?? 1) - (a.contentVersion ?? 1);
  });

  // Step 5: Task context boost — ensure task-specific chunk in results if taskContext provided
  let results = reranked.slice(0, 5);
  if (input.taskContext) {
    const taskChunkKey = `CHUNK_${input.taskContext}_${input.personaKey}`;
    const hasTaskChunk = results.some(r => r.chunkKey === taskChunkKey);
    if (!hasTaskChunk) {
      const taskChunk = reranked.find(r => r.chunkKey === taskChunkKey);
      if (taskChunk) {
        results = [taskChunk, ...results.slice(0, 4)];
      }
    }
  }

  logger.info(
    { query: input.query, personaKey: input.personaKey, chunkCount: results.length },
    'RAG retrieval complete',
  );

  // Step 6: Return 3–5 chunks
  return results.map(r => ({
    chunkKey: r.chunkKey,
    content: r.content,
    sourceUrl: r.sourceUrl,
    sourceName: r.sourceName,
    similarityScore: Number(r.similarity),
  }));
}
```
