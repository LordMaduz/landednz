## Exact Implementation

**`level1.ts` additions:**

```typescript
// OUT_OF_SCOPE_IMMIGRATION patterns — case-insensitive substring match
// Targets immigration enforcement/legal matters, not legitimate settlement questions
const OUT_OF_SCOPE_IMMIGRATION_PATTERNS: string[] = [
  "visa breach",
  "visa condition violated",
  "deportation",
  "overstayed",
  "visa appeal",
  "immigration lawyer",
  "visa cancelled",
  "unlawful stay",
];

// Empathetic redirect — model must NOT answer the immigration legal question
// Injected into Component 3 of system prompt via buildChatSystemPrompt()
const OUT_OF_SCOPE_IMMIGRATION_AUGMENTATION =
  `OUT OF SCOPE — REDIRECT ONLY:\n` +
  `This question involves immigration legal matters that are outside what I can ` +
  `help with. Acknowledge the concern with empathy, then direct the user to ` +
  `immigration.govt.nz/find-an-adviser where they can find a licensed immigration ` +
  `adviser. Do not attempt to answer the immigration legal question.`;
```

**`runLevel1Guardrail()` branch (after PROMPT_INJECTION check):**

```typescript
if (OUT_OF_SCOPE_IMMIGRATION_PATTERNS.some(pattern => lower.includes(pattern))) {
  return {
    triggered: true,
    flags: ['OUT_OF_SCOPE_IMMIGRATION'],
    augmentations: [OUT_OF_SCOPE_IMMIGRATION_AUGMENTATION],
  };
}
```

**`level1.test.ts` new describe block:**

```typescript
describe('runLevel1Guardrail — OUT_OF_SCOPE_IMMIGRATION', () => {
  test('exact AC test input fires OUT_OF_SCOPE_IMMIGRATION', () => {
    const result = runLevel1Guardrail("my visa was breached");
    expect(result.triggered).toBe(true);
    expect(result.flags).toContain('OUT_OF_SCOPE_IMMIGRATION');
    expect(result.augmentations).toHaveLength(1);
    expect(result.augmentations[0]).toContain('immigration.govt.nz/find-an-adviser');
  });

  test('"I think I have overstayed" fires OUT_OF_SCOPE_IMMIGRATION', () => {
    const result = runLevel1Guardrail("I think I have overstayed my visa");
    expect(result.triggered).toBe(true);
    expect(result.flags).toContain('OUT_OF_SCOPE_IMMIGRATION');
  });

  test('"I need an immigration lawyer" fires OUT_OF_SCOPE_IMMIGRATION', () => {
    const result = runLevel1Guardrail("I need an immigration lawyer urgently");
    expect(result.triggered).toBe(true);
    expect(result.flags).toContain('OUT_OF_SCOPE_IMMIGRATION');
  });

  test('case-insensitive — "VISA BREACH" fires', () => {
    const result = runLevel1Guardrail("VISA BREACH — what do I do?");
    expect(result.triggered).toBe(true);
    expect(result.flags).toContain('OUT_OF_SCOPE_IMMIGRATION');
  });

  test('in-scope guard — "visa conditions for working" does NOT fire', () => {
    const result = runLevel1Guardrail("What are my visa conditions for working?");
    expect(result.triggered).toBe(false);
    expect(result.flags).toHaveLength(0);
  });

  test('normal message does not fire OUT_OF_SCOPE_IMMIGRATION', () => {
    const result = runLevel1Guardrail("How do I open a bank account?");
    expect(result.triggered).toBe(false);
  });
});
```
